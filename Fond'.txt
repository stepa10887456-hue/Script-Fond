local Tab = Window:CreateTab("Visual") -- Title, Image

local Section = Tab:CreateSection("Esp")

local Button = Tab:CreateButton({
 Name = "Esp Doors",
 Callback = function()
 -- Made by mrpopcat14
local RunService = game:GetService("RunService")
local CollectionService = game:GetService("CollectionService")

-- Таблица для отслеживания выделенных объектов
local highlightedObjects = {}

local function handleNewObject(obj)
 if obj:IsA('Part') and obj.Parent and obj.Parent:IsA('Model') and obj.Родитель.Name == 'Door' then
 -- Проверить, не выделен ли уже объект
 if not highlightedObjects[obj] then
 if not obj.Parent:FindFirstChildOfClass('BillboardGui') then
 local BillboardGui = Instance.new('BillboardGui')
 local TextLabel = Instance.new('TextLabel')

 BillboardGui.Parent = obj.Родитель
 BillboardGui.AlwaysOnTop = true
 BillboardGui.Size = UDim2.new(0, 50, 0, 50)
 BillboardGui.StudsOffset = Vector3.new(0, 2, 0)

 TextLabel.Parent = BillboardGui
 TextLabel.BackgroundTransparency = 1
 TextLabel.Size = UDim2.new(1, 0, 1, 0)
 TextLabel.Text = "Door" -- Отображение текста
 TextLabel.TextScaled = true
 TextLabel.TextStrokeTransparency = 0.5 -- Обводка текста для видимости

 -- Убрана проверка начального поворота и связанная с ней логика

 local function updateColors()
 local hue = tick() % 5 / 5 
 local color = Color3.fromHSV(hue, 1, 1)
 TextLabel.TextColor3 = color
 end

 -- Продолжаем обновлять цвета без проверки поворота
 RunService.RenderStepped:Connect(updateColors)

 highlightedObjects[obj] = true
 end
 end
 end
end

-- Функция для пакетной обработки всех существующих объектов
local function handleExistingObjects()
 for _, obj in ipairs(game.Workspace:GetDescendants()) do
 handleNewObject(obj)
 end
end

игра.Рабочее место.ChildAdded:Connect(function(child)
 handleNewObject(child)
end)

local lastCheck = 0
RunService.Heartbeat:Connect(function(deltaTime)
 lastCheck = lastCheck + deltaTime
 if lastCheck >= 0.5 then
 handleExistingObjects()
 lastCheck = 0
 end
end)

handleExistingObjects()
 конец,
})
